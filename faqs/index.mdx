            ...section,
            questions: [
              ...section.questions.filter((item) => !commonQuestions.includes(item)),

    : sections;

    .map((section) => ({
      ...section,
      questions: section.questions.filter((item) =>
        `${section.title} ${item.question} ${item.answer} ${item.keywords}`
          .toLowerCase()
          .includes(searchTerm),
      ),
    }))
    .filter((section) => section.questions.length > 0);
  const resultCount = filteredSections.reduce(
    (count, section) => count + section.questions.length,
    0,
  );
  const roleLabel =
    role === "educator" ? "Educator" : role === "parent" ? "Parent" : "Student";

  return (
